2.1. Kiến trúc Cơ sở Dữ liệu NoSQL trên MongoDB Atlas

Hệ thống sử dụng MongoDB Atlas kết hợp với Mongoose ODM (v8.3+) để xây dựng mô hình dữ liệu quan hệ hướng tài liệu. Cấu trúc Schema được thiết kế đồng bộ với hệ thống PRM393, hỗ trợ đầy đủ các trường kiểm soát tồn kho chi nhánh, mã giảm giá, đánh giá và tin nhắn CSKH quầy.

2.2. Chi tiết 7 Schemas Cốt lõi (Mongoose Models)

1. Model Người dùng (models/User.js)

const mongoose = require('mongoose');
const bcrypt = require('bcryptjs');

const userSchema = new mongoose.Schema(
  {
    fullName: { type: String, required: true, trim: true },
    email: { type: String, required: true, unique: true, lowercase: true, trim: true },
    password: { type: String, required: true },
    phone: { type: String, required: true, trim: true },
    avatar: {
      type: String,
      default: 'https://images.unsplash.com/photo-1535713875002-d1d0cf377fde?w=200'
    },
    role: {
      type: String,
      enum: ['user', 'admin'],
      default: 'user'
    },
    favoriteProducts: [
      {
        type: mongoose.Schema.Types.ObjectId,
        ref: 'Product'
      }
    ]
  },
  { timestamps: true }
);

// Mongoose Pre-save Hook: Tự động băm mật khẩu bằng Bcrypt trước khi lưu vào MongoDB
userSchema.pre('save', async function () {
  if (!this.isModified('password')) return;
  const salt = await bcrypt.genSalt(10);
  this.password = await bcrypt.hash(this.password, salt);
});

// So khớp mật khẩu đăng nhập
userSchema.methods.comparePassword = async function (candidatePassword) {
  return bcrypt.compare(candidatePassword, this.password);
};

// Tự động loại bỏ password khi trả dữ liệu về Client
userSchema.methods.toJSON = function () {
  const user = this.toObject();
  delete user.password;
  return user;
};

module.exports = mongoose.model('User', userSchema);

PHẦN 4: XÁC THỰC JWT, BCRYPT, VALIDATION MATRIX & XỬ LÝ LỖI TOÀN CỤC

4.1. Cơ chế Bảo mật Xác thực JWT & Mã hóa Mật khẩu Bcrypt

Hệ thống bảo vệ tài nguyên API bằng giao thức JSON Web Token (JWT) theo tiêu chuẩn Stateless Authentication:

  • Mã hóa mật khẩu: Mật khẩu người dùng được băm 1 chiều bằng thuật toán bcryptjs với độ phức tạp saltRounds = 10. Chuỗi băm sinh ra có độ dài 60 ký tự, chống hoàn toàn kỹ thuật tấn công từ điển và Rainbow Table.
  • Cấu trúc JWT Payload: Chỉ chứa định danh tối thiểu { id: user._id, role: user.role }. Tuyệt đối không lưu trữ mật khẩu, email hoặc thông tin nhạy cảm bên trong Payload.
  • Thời hạn hiệu lực: Token được cấu hình hết hạn sau 7 ngày (expiresIn: '7d').
  • Header truyền tải: Phía Client gửi Token qua HTTP Header theo định dạng chuẩn: Authorization: Bearer <token>.

Middleware Xác thực Token & Phân quyền (middlewares/authMiddleware.js)

const jwt = require('jsonwebtoken');
const User = require('../models/User');

// Giải mã và xác thực token từ header Authorization
exports.verifyToken = async (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({
        success: false,
        message: 'Yêu cầu đăng nhập. Thiếu Bearer Token trong header!'
      });
    }

    const token = authHeader.split(' ')[1];
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    const user = await User.findById(decoded.id);
    if (!user) {
      return res.status(401).json({ success: false, message: 'Người dùng không tồn tại' });
    }

    req.user = user; // Gắn thông tin user vào Request để các controller tiếp theo sử dụng
    next();
  } catch (error) {
    return res.status(401).json({
      success: false,
      message: error.name === 'TokenExpiredError' ? 'Token đã hết hạn' : 'Token không hợp lệ'
    });
  }
};

// Kiểm tra Role người dùng để phân quyền truy cập
exports.checkRole = (...roles) => {
  return (req, res, next) => {
    if (!req.user || !roles.includes(req.user.role)) {
      return res.status(403).json({
        success: false,
        message: 'Bạn không có quyền thực hiện hành động này!'
      });
    }
    next();
  };
};

exports.authorize = exports.checkRole;

4.3. Middleware Bắt Lỗi Tập Trung Toàn Cục (middlewares/errorMiddleware.js)

Để ứng dụng không bao giờ bị dừng đột ngột (crash) và luôn phản hồi mã HTTP chuẩn xác, mọi ngoại lệ trong Express đều được chuyển tiếp qua hàm next(error) và xử lý tại errorMiddleware.js: 

// 1. Middleware bắt các route không tồn tại (404 Not Found)
exports.notFound = (req, res, next) => {
  const error = new Error(`Resource not found: ${req.originalUrl}`);
  res.status(404);
  next(error);
};

// 2. Global Error Handler: Phân loại và chuyển đổi mã lỗi
exports.errorHandler = (err, req, res, next) => {
  let statusCode = res.statusCode === 200 ? 500 : res.statusCode;
  let message = err.message;

  // Bắt lỗi truyền ID sai định dạng MongoDB ObjectId (CastError)
  if (err.name === 'CastError' && err.kind === 'ObjectId') {
    statusCode = 400;
    message = 'Invalid ID format';
  }
  // Bắt lỗi trùng lặp trường unique trong MongoDB (Ví dụ trùng Email, trùng Category code, trùng Voucher code)
  else if (err.code === 11000) {
    statusCode = 400;
    const field = err.keyValue ? Object.keys(err.keyValue)[0] : 'Field';
    message = `${field} already exists`;
  }
  // Bắt lỗi Schema Validation của Mongoose
  else if (err.name === 'ValidationError') {
    statusCode = 400;
    message = Object.values(err.errors).map(e => e.message).join(', ');
  }

  res.status(statusCode).json({
    success: false,
    message,
    ...(process.env.NODE_ENV !== 'production' && { stack: err.stack })
  });
};
Nguyên tắc sống còn trong Express.js: Middleware xử lý lỗi (errorHandler) bắt buộc phải nhận đủ 4 tham số (err, req, res, next) và phải được đăng ký ở cuối cùng trong server.js, ngay sau tất cả các khai báo route. Nếu đặt sai thứ tự hoặc thiếu tham số err, Express sẽ coi đó là middleware thông thường và không thể chặn được các exception nảy sinh từ tầng Controller.

4.4. Cơ Chế Bảo Mật Kết Nối HTTPS, Chứng Chỉ SSL/TLS & Helmet Security Headers (Slot 11)

Nhằm bảo vệ tuyệt đối luồng thanh toán và dữ liệu nhạy cảm của khách hàng theo chuẩn bài giảng Slot 11, hệ thống FreshBite triển khai kiến trúc truyền thông bảo mật:

  • Nguyên lý HTTPS & Bắt tay SSL/TLS Handshake: Quá trình kết nối kết hợp giữa Mã hóa bất đối xứng (Asymmetric Encryption) để xác thực danh tính máy chủ và trao đổi khóa bí mật dùng chung (Session Key), sau đó chuyển sang Mã hóa đối xứng (Symmetric Encryption - AES) để mã hóa luồng dữ liệu HTTP với tốc độ cao, tránh làm nghẽn CPU của Node.js.
  • Khởi tạo chứng chỉ tự ký (Self-signed Certificate) với OpenSSL: 
    openssl req -x509 -newkey rsa:2048 -nodes -sha256 -keyout server.key -out server.cert -days 365
  • Khởi tạo HTTPS Server trong Node.js (Slot 11): 
    const https = require('https');
const fs = require('fs');
const path = require('path');

const sslOptions = {
  key: fs.readFileSync(path.join(__dirname, 'server.key')),

};

// Lắng nghe cổng HTTPS 443 hoặc cổng dev bảo mật

});
  • Bảo vệ HTTP Header với Helmet: Sử dụng app.use(helmet()) nhằm chặn đứng các cuộc tấn công phổ biến: Clickjacking (X-Frame-Options: SAMEORIGIN), MIME Sniffing (X-Content-Type-Options: nosniff), và bắt buộc kết nối an toàn qua HSTS.
  • Kiến trúc SSO & OAuth 2.0 (Google Login): Sẵn sàng tích hợp ủy quyền đăng nhập qua 4 thành phần chuẩn: Resource Owner (Khách hàng), Client (FreshBite App), Authorization Server (Google Auth Server) và Resource Server (Google User Profile API).

5.1. Mô hình Kiến trúc MVC & Middleware Pipeline

Hệ thống tuân thủ nghiêm ngặt mô hình phân tầng MVC (Model - View - Controller) mở rộng cho kiến trúc RESTful API:

File server.js là trái tim của hệ thống Backend, kiểm soát thứ tự thực thi tuần tự của các Middleware: 

require('dotenv').config();
const express = require('express');
const cors = require('cors');
const helmet = require('helmet');
const morgan = require('morgan');
const path = require('path');
const connectDB = require('./config/db');
const { notFound, errorHandler } = require('./middlewares/errorMiddleware');

const app = express();

// 1. Bảo mật HTTP Headers với Helmet
app.use(helmet({ crossOriginResourcePolicy: { policy: 'cross-origin' } }));

// 2. Cho phép Cross-Origin Resource Sharing (CORS) cho Mobile App & Web kết nối
app.use(cors());

// 3. Ghi log HTTP requests khi chạy môi trường development
if (process.env.NODE_ENV === 'development') {
  app.use(morgan('dev'));
}

// 4. Phân tích nội dung JSON và URL-encoded trong Request Body
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// 5. Cung cấp file tĩnh từ thư mục uploads
app.use('/uploads', express.static(path.join(__dirname, 'uploads')));

// 6. Root & Health Check Endpoints
app.get('/', (req, res) => res.json({ status: 'ok', message: 'FreshBite API running' }));
app.get('/api/health', (req, res) => res.json({ status: 'ok', timestamp: new Date().toISOString() }));

// 7. Khai báo các Module Routes
app.use('/api/auth', authRoutes);
app.use('/api/categories', categoryRoutes);
app.use('/api/products', productRoutes);
app.use('/api/orders', orderRoutes);
app.use('/api/branches', branchRoutes);
app.use('/api/vouchers', voucherRoutes);
app.use('/api/reviews', reviewRoutes);
app.use('/api/chats', chatRoutes);
app.use('/api/reports', reportRoutes);

// 8. Bắt buộc đặt ở cuối cùng: Bắt route không tồn tại (404) và Xử lý lỗi toàn cục

app.use(errorHandler);

const PORT = process.env.PORT || 5000;
let server;

// Kết nối MongoDB thành công mới mở Port lắng nghe kết nối
connectDB().then(() => {
  server = app.listen(PORT, () => {
    console.log(`Server running in ${process.env.NODE_ENV || 'development'} mode on port ${PORT}`);
  });
});

// Chống dừng ứng dụng khi gặp Unhandled Promise Rejection
process.on('unhandledRejection', (err) => {
  console.error(`Unhandled Rejection: ${err.message}`);
  if (server) server.close(() => process.exit(1));
  else process.exit(1);
});

2. Dành cho: Nguyễn Chí Vương (API Architect / Auth & Security Lead - 20%)

Câu hỏi 1: JWT là gì? Payload của em chứa những thông tin gì? Tại sao không được lưu password vào JWT?
Trả lời: JWT (JSON Web Token) là chuẩn mở (RFC 7519) định nghĩa cách truyền tải thông tin an toàn giữa các bên dưới dạng JSON object. JWT gồm 3 phần: Header, Payload và Signature. Trong dự án, Payload em chỉ lưu định danh tối thiểu gồm: { id: user._id, role: user.role }. Tuyệt đối không lưu mật khẩu vào Payload vì Payload chỉ được mã hóa Base64Url chứ không hề được mã hóa bí mật, bất kỳ ai cầm token đều có thể giải mã Base64 ra đọc được Payload. Token chỉ an toàn ở tính toàn vẹn (Integrity) nhờ Signature được ký với JWT_SECRET trên Server.
Câu hỏi 2: Trong Postman Collection, em làm thế nào để request tạo đơn hàng tự động có Bearer Token mà không cần copy paste tay?
Trả lời: Tại request Login trong Postman, tại tab Tests, em viết đoạn mã JavaScript: pm.collectionVariables.set("token", pm.response.json().token);. Khi đăng nhập thành công, Postman tự động bắt lấy chuỗi token và gán vào biến Collection có tên là token. Tại các request tiếp theo cần xác thực, ở tab Authorization, em chọn loại Bearer Token và truyền giá trị là {{token}}. Nhờ đó mọi request tự động kế thừa token vừa đăng nhập.
Câu hỏi 3: Phương thức toJSON trong User Schema hoạt động thế nào? Module Favorites được lưu trữ ra sao?
Trả lời: Khi Document được chuyển sang JSON (như khi gọi res.json(user)), Mongoose sẽ tự gọi hàm toJSON(). Em override hàm này bằng: userSchema.methods.toJSON = function() { const u = this.toObject(); delete u.password; return u; } để đảm bảo trường password luôn bị loại bỏ trước khi trả về Client. Về Favorites, trong User Schema em có mảng favoriteProducts: [{ type: ObjectId, ref: 'Product' }]. Trả lời: Vì Client là môi trường không an toàn, kẻ gian hoàn toàn có thể can thiệp gói tin để sửa giá từ 45.000đ xuống 1.000đ. Do đó, server chỉ nhận productId, size và quantity. Câu hỏi 2: Giải thích cơ chế Mongoose Population? Khác gì với $lookup trong MongoDB Aggregation?
Trả lời: Mongoose populate() là cơ chế giúp thay thế trường ObjectId tham chiếu bằng Document thực tế từ collection khác. Bên dưới Mongoose sẽ thực hiện 1 truy vấn riêng biệt thứ hai bằng lệnh find({ _id: { $in: ids } }) sau đó ghép dữ liệu lại trên bộ nhớ Node.js. Còn $lookup là toán tử thực hiện trực tiếp bên trong Database Engine thông qua Aggregation Pipeline (tương đương Left Outer Join của SQL). Dùng populate() giúp mã nguồn trong Controller ngắn gọn, dễ bảo trì và đúng cấu trúc chuẩn của Mongoose.

Khi controller gọi Order.findById("my-orders"), Mongoose quăng lỗi CastError vì chuỗi này không phải là ObjectId 24 ký tự hex hợp lệ. Câu hỏi 1: Tính năng kiểm soát món tạm hết hàng tại chi nhánh (outOfStockProductIds) hoạt động thế nào?
Trả lời: Mỗi chi nhánh có một danh sách tồn kho riêng. Trong Model Branch, em có mảng outOfStockProductIds chứa ID các món tạm hết tại chi nhánh đó. Khi Barista phát hiện hết nguyên liệu làm Cà phê Muối tại quầy FPT City, họ gọi API PATCH /api/branches/:id/stock để thêm ID món đó vào mảng. Khi khách đặt đơn hàng tại chi nhánh FPT City, hàm createOrder sẽ kiểm tra: nếu món khách chọn nằm trong outOfStockProductIds của chi nhánh, server trả về mã lỗi 400 Bad Request kèm thông báo món đã tạm hết tại quầy này, đồng bộ hoàn toàn với Tab 2 của PRM393.
Câu hỏi 2: Module Voucher kiểm tra những điều kiện gì trước khi áp dụng giảm giá?
Trả lời: Trong voucherController.js, khi khách gửi mã giảm giá lên, server kiểm tra tuần tự 4 điều kiện: 1. Mã voucher có tồn tại và isActive = true hay không; 2. Ngày hiện tại có nằm trong khoảng từ startDate đến endDate hay không; 3. Tổng tiền đơn hàng có đạt mức tối thiểu minOrderAmount hay không; 4. Tính số tiền giảm: nếu là PERCENT thì tính theo % nhưng không vượt quá maxDiscount; nếu là FIXED thì trừ trực tiếp số tiền cố định.
Câu hỏi 3: Tại sao Middleware errorHandler phải nhận đủ 4 tham số và đặt ở cuối cùng của file server.js?
Trả lời: Express nhận diện middleware xử lý lỗi dựa trên số lượng tham số khai báo của hàm (fn.length === 4). Nếu chỉ khai báo 3 tham số (err, req, res), Express sẽ coi đó là middleware thông thường. Nó phải được đặt ở cuối cùng của server.js vì middleware chỉ bắt được lỗi của các route được khai báo phía trên nó khi các controller gọi next(error).

5. Dành cho: Lê Tự Nhật Thạch (Uploads, Seed Data & Web Admin SSR - 20%)

Câu hỏi 1: Thư viện Multer hoạt động ra sao? Làm sao để client tải được ảnh tĩnh từ thư mục /uploads?
Trả lời: Multer là middleware chuyên phân tích định dạng multipart/form-data. Em cấu hình diskStorage để lưu file vào thư mục uploads/ và đổi tên file bằng Date.now() + path.extname(file.originalname) để tránh bị trùng tên file. Để client tải được ảnh, trong server.js em cấu hình static middleware: app.use('/uploads', express.static(path.join(__dirname, 'uploads')));. Nhờ đó Client có thể xem trực tiếp ảnh qua URL: http://localhost:5000/uploads/ten-anh.jpg.
Câu hỏi 2: Giao diện SSR bằng EJS khác gì so với Client-Side Rendering (CSR) trên React Native hay Flutter?
Trả lời: Với EJS (Server-Side Rendering), mã HTML hoàn chỉnh được tạo ra và ghép dữ liệu ngay trên máy chủ Node.js trước khi gửi về cho trình duyệt hiển thị. Ưu điểm là tải trang nhanh, không phụ thuộc vào sức mạnh xử lý của thiết bị client, phù hợp cho màn hình quản lý quầy Barista tại cửa hàng. Còn CSR (Client-Side Rendering) như React Native hay Flutter thì Server chỉ trả về dữ liệu thô dạng JSON, sau đó điện thoại của khách hàng mới tự vẽ giao diện và gắn dữ liệu vào từng Widget.
Câu hỏi 3: Trong script seed tài khoản (userSeed.js), tại sao em dùng User.create() thay vì insertMany()?
Trả lời: Vì trong User Model, em có cài đặt hook userSchema.pre('save', ...) để tự động băm mật khẩu bằng Bcrypt. Hàm User.create() khởi tạo từng document và gọi hàm save() nên hook băm mật khẩu sẽ được kích hoạt bình thường. Nếu dùng User.insertMany(), MongoDB sẽ chèn trực tiếp dữ liệu thô vào database mà bỏ qua hook save(), khiến mật khẩu lưu dưới dạng plain text và người dùng không thể đăng nhập được.
Câu hỏi 4: Phương thức res.download() hoạt động ra sao? Khác gì với res.sendFile() và static files? (Slot 12)
Trả lời: res.download(path, [filename]) tự động cấu hình HTTP Header Content-Disposition: attachment; filename="...", yêu cầu trình duyệt hoặc client di động kích hoạt hộp thoại tải file về máy thay vì hiển thị trực tiếp (inline display) như res.sendFile(). Trong FreshBite, hàm này được ứng dụng để xuất file hóa đơn bán lẻ (/api/orders/:id/invoice) và file CSV đối soát ca (/api/reports/shift/export).
Câu hỏi 5: Giao thức HTTPS và bắt tay SSL/TLS hoạt động thế nào trên Node.js? Tại sao lại dùng cả mã hóa đối xứng và bất đối xứng? (Slot 11)
Trả lời: Bắt tay TLS Handshake kết hợp 2 giai đoạn: (1) Mã hóa bất đối xứng (Public/Private Key) ở giai đoạn đầu để xác thực danh tính server và trao đổi an toàn một khóa phiên bí mật (Session Key); (2) Sau đó toàn bộ dữ liệu HTTP được mã hóa bằng Mã hóa đối xứng (AES) sử dụng khóa phiên đó vì mã hóa đối xứng có tốc độ xử lý nhanh hơn hàng nghìn lần, không gây quá tải CPU của máy chủ Node.js. Trên môi trường local, ta tạo self-signed cert bằng OpenSSL và dùng module https.createServer({ key, cert }, app).
Câu hỏi 6: So sánh kiến trúc Express.js và NestJS? Nếu mở rộng hệ thống FreshBite toàn quốc thì tại sao nên chuyển sang NestJS? (Slot 17-19)
Trả lời: Express là micro-framework tối giản và linh hoạt, rất phù hợp để hiểu sâu cơ chế HTTP và xây dựng MVP nhanh. Tuy nhiên, khi mở rộng quy mô lớn (Enterprise) với chuỗi hàng trăm chi nhánh, NestJS vượt trội hơn nhờ áp dụng kiến trúc Modular chuẩn hóa (Modules, Controllers, Providers/Services), tích hợp sẵn TypeScript, cơ chế Dependency Injection (DI) và DTO validation. Giúp phân chia công việc nhóm rõ ràng, mã nguồn dễ bảo trì, testable và loại trừ triệt để lỗi runtime.
